Generics

Functions, structs, enums and impl blocks can be generic.

Bounds

Bounds work inline and in where clauses. A generic type without the bound it needs fails the check, as in rustc. Operator traits from std::ops, such as Add, work.

generics.rs
use std::fmt;
use std::ops::Add;

struct Stack<T> {
    items: Vec<T>,
}

impl<T: Clone> Stack<T> {
    fn new() -> Self {
        Stack { items: Vec::new() }
    }

    fn push(&mut self, item: T) {
        self.items.push(item);
    }

    fn top(&self) -> Option<T> {
        self.items.last().cloned()
    }
}

#[derive(Debug, Clone, Copy, PartialEq)]
struct Meters(f64);

impl Add for Meters {
    type Output = Meters;

    fn add(self, other: Meters) -> Meters {
        Meters(self.0 + other.0)
    }
}

impl fmt::Display for Meters {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        write!(f, "{} m", self.0)
    }
}

fn largest<T: PartialOrd + Copy>(items: &[T]) -> T {
    let mut max = items[0];
    for &item in items {
        if item > max {
            max = item;
        }
    }
    max
}

fn main() {
    let mut stack = Stack::new();
    stack.push(Meters(1.5) + Meters(2.0));
    println!("{}", stack.top().unwrap());
    println!("{}", largest(&[3, 9, 2]));
}
rawr run generics.rs3.5 m 9

Trait objects

Box<dyn Trait> holds any type that implements the trait. &dyn Trait arguments aren’t supported yet.

trait Shape {
    fn area(&self) -> f64;
}

struct Square(f64);
struct Circle(f64);

impl Shape for Square {
    fn area(&self) -> f64 {
        self.0 * self.0
    }
}

impl Shape for Circle {
    fn area(&self) -> f64 {
        3.14 * self.0 * self.0
    }
}

let shapes: Vec<Box<dyn Shape>> = vec![Box::new(Square(2.0)), Box::new(Circle(1.0))];
let total: f64 = shapes.iter().map(|s| s.area()).sum();
println!("{total}");
rawr run shapes.rs7.140000000000001

Iterator

A type that implements Iterator gets map, filter, collect and the other adapters.

struct Countdown(u32);

impl Iterator for Countdown {
    type Item = u32;

    fn next(&mut self) -> Option<u32> {
        if self.0 == 0 {
            return None;
        }
        self.0 -= 1;
        Some(self.0)
    }
}

let v: Vec<u32> = Countdown(3).collect(); // [2, 1, 0]
println!("{v:?}");
rawr run countdown.rs[2, 1, 0]

FromIterator

A type that implements FromIterator can be built by collect().

struct Bag(Vec<i64>);

impl FromIterator<i64> for Bag {
    fn from_iter<I: IntoIterator<Item = i64>>(iter: I) -> Self {
        Bag(iter.into_iter().collect())
    }
}

let bag: Bag = (1..4).collect();
println!("{:?}", bag.0);
rawr run bag.rs[1, 2, 3]
